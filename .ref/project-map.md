```

│   ├── ai-tools/            # 22 AI tool modules (barrel: server/ai-tools.ts)
│   ├── export/              # KiCad/Eagle/SPICE/Gerber/drill/PnP/firmware-scaffold generators
│   ├── firmware-runtime/    # firmware exec runtime
│   ├── storage/             # PartsStorage, SupplyChainStorage, BomTemplateStorage
│   └── lib/                 # vault-search, lru-cache, helpers
├── shared/                  # LEGACY shared layer (Drizzle+Zod schema, DRC, diffs, parts utils)
├── packages/                # THE ENGINE redesign — 16 @protopulse/* npm workspaces
│   ├── graph/               # THE PRODUCT: op-log design graph, .ppx format, branch/diff/merge
│   ├── parts/  erc/  export/  drc/  route/        # part model, electrical rules, fab exports, DRC, routing
│   ├── sim/  emu/  cosim/                          # ngspice-WASM, MCU cores, firmware↔analog loop
│   ├── review/  relay/  renderer/  ai/  content/   # review decks, sync relay, WebGL2, AI crew, content schemas
│   ├── cli/                 # protopulse check/export/import-legacy
│   └── app/                 # the new editor (schematic+PCB, panels for the whole crew)
├── tools/golden/            # CONTRACT: golden op-logs → byte-exact export fixtures
├── content/                 # fab decks, concepts wiki, review decks, catalog, tracks, puzzles
├── docs/                    # vision/ (FROZEN), adr/, plans/, audits/, MASTER_BACKLOG.md, CHANGELOG.md
```
