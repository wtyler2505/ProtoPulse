// Blink + LDR sensor read — ProtoPulse demo
#define LED_PIN 13
#define SENSOR_PIN A0
const long BAUD = 9600;
int reading = 0;
void setup() {
pinMode(LED_PIN, OUTPUT);
pinMode(SENSOR_PIN, INPUT);
Serial.begin(BAUD);
Serial.println("ProtoPulse ready");
}
void loop() {
reading = analogRead(SENSOR_PIN);
float volts = reading * (5.0 / 1023.0);
digitalWrite(LED_PIN, HIGH);
delay(250);
digitalWrite(LED_PIN, LOW);
delay(250);
Serial.print("Sensor: ");
Serial.print(reading);
Serial.print(" V=");
Serial.println(volts);
